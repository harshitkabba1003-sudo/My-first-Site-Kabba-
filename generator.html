document.getElementById('generateBtn').addEventListener('click', async () => { const urlInput = document.getElementById('youtubeUrl').value; const resultContainer = document.getElementById('resultContainer'); const strategyOutput = document.getElementById('strategyOutput'); const downloadBtn = document.getElementById('downloadBtn'); if (!urlInput) { alert("Please enter a valid YouTube URL!"); return; } // Show loading state, hide previous results resultContainer.style.display = 'block'; strategyOutput.innerText = "Analyzing video and generating your viral strategy pack... Please wait."; downloadBtn.style.display = 'none'; try { const response = await fetch("https://harshitkabba-viral-shorts-backend.hf.space/api/v1/run/1efa6577-ef8b-44af-8b60-a8d8fcbdcdf4", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ input_value: urlInput, input_type: "chat", output_type: "chat", tweaks: {} // Empty because no email or fields need overriding now! }) }); const data = await response.json(); // Extract the generated text from Langflow's standardized response const generatedStrategy = data.outputs[0].outputs[0].results.message.text; if (generatedStrategy) { strategyOutput.innerText = generatedStrategy; downloadBtn.style.display = 'inline-block'; // Reveal download button // Configure the download button logic downloadBtn.onclick = () => { const blob = new Blob([generatedStrategy], { type: 'text/plain' }); const url = window.URL.createObjectURL(blob); const a = document.createElement('a'); a.href = url; a.download = 'AI_Viral_Shorts_Strategy.txt'; document.body.appendChild(a); a.click(); window.URL.revokeObjectURL(url); document.body.removeChild(a); }; } else { strategyOutput.innerText = "Error: Couldn't parse response. Please check your Langflow model settings."; } } catch (error) { console.error(error); strategyOutput.innerText = "Langflow Server Error: " + error.message; } });
